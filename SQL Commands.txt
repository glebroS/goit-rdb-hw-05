USE hw3_dql;

-- ==========================================
-- ЗАВДАННЯ 1: Вкладений запит в SELECT
-- Відображає order_details та відповідний customer_id з orders
-- ==========================================
SELECT 
    od.*, 
    (SELECT customer_id FROM orders o WHERE o.id = od.order_id) AS customer_id
FROM order_details od;


-- ==========================================
-- ЗАВДАННЯ 2: Вкладений запит в WHERE
-- Відфільтрувати order_details так, щоб shipper_id = 3
-- ==========================================
SELECT * FROM order_details 
WHERE order_id IN (
    SELECT id 
    FROM orders 
    WHERE shipper_id = 3
);


-- ==========================================
-- ЗАВДАННЯ 3: Вкладений запит в FROM
-- quantity > 10, знайти середнє значення quantity, згрупувати за order_id
-- ==========================================
SELECT 
    order_id, 
    AVG(quantity) AS avg_quantity 
FROM (
    SELECT * FROM order_details 
    WHERE quantity > 10
) AS filtered_orders 
GROUP BY order_id;


-- ==========================================
-- ЗАВДАННЯ 4: Використання оператора WITH (CTE)
-- ==========================================
WITH temp AS (
    SELECT * FROM order_details 
    WHERE quantity > 10
)
SELECT 
    order_id, 
    AVG(quantity) AS avg_quantity 
FROM temp 
GROUP BY order_id;


-- ==========================================
-- ЗАВДАННЯ 5: Створення власної функції
-- Ділення двох FLOAT параметрів
-- ==========================================
DROP FUNCTION IF EXISTS divide_float;

DELIMITER //

CREATE FUNCTION divide_float(num1 FLOAT, num2 FLOAT)
RETURNS FLOAT
DETERMINISTIC
BEGIN
    IF num2 = 0 THEN
        RETURN NULL; -- Захист від ділення на нуль
    END IF;
    RETURN num1 / num2;
END //

DELIMITER ;

-- Застосування функції до атрибута quantity (ділимо на довільне число 2.5)
SELECT 
    id,
    order_id, 
    product_id, 
    quantity, 
    divide_float(quantity, 2.5) AS divided_quantity 
FROM order_details;
